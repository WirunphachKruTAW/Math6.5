    const form = document.getElementById("lookup-form");
    const idInput = document.getElementById("student-id");
    const message = document.getElementById("lookup-message");
    const lookupSection = document.getElementById("lookup-section");
    const resultSection = document.getElementById("result-section");
    const resultName = document.getElementById("result-name");
    const resultId = document.getElementById("result-id");
    const resultScore = document.getElementById("result-score");

    function clearResult() {
      resultName.textContent = "";
      resultId.textContent = "";
      resultScore.textContent = "";
    }

    form.addEventListener("submit", function(event) {
      event.preventDefault();
      clearResult();
      resultSection.hidden = true;
      const id = idInput.value.trim();
      const student = students.find(item => item.id === id);
      if (!student) {
        message.textContent = "ไม่พบข้อมูล กรุณาตรวจสอบเลขประจำตัวนักเรียนอีกครั้ง";
        message.hidden = false;
        idInput.setAttribute("aria-invalid", "true");
        return;
      }
      message.hidden = true;
      message.textContent = "";
      idInput.removeAttribute("aria-invalid");
      resultName.textContent = student.name;
      resultId.textContent = student.id;
      resultScore.textContent = String(student.score);
      lookupSection.hidden = true;
      resultSection.hidden = false;
      document.getElementById("results-title").setAttribute("tabindex", "-1");
      document.getElementById("results-title").focus();
    });

    document.getElementById("logout-button").addEventListener("click", function() {
      clearResult();
      idInput.value = "";
      idInput.removeAttribute("aria-invalid");
      message.hidden = true;
      message.textContent = "";
      resultSection.hidden = true;
      lookupSection.hidden = false;
      idInput.focus();
    });

    lucide.createIcons();
